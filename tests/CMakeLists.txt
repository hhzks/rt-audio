function(rt_add_test name)
    cmake_parse_arguments(PARSE_ARGV 1 RT "NO_CONTRACT_HANDLER;ALLOW_VIOLATIONS" "" "")
    add_executable(${name} ${name}.cpp)
    target_link_libraries(${name} PRIVATE rt_engine rt_dsp rt_core Catch2::Catch2WithMain)
    rt_set_warnings(${name})
    if(RT_NO_CONTRACT_HANDLER)
        catch_discover_tests(${name} TEST_PREFIX "${name}: ")
    elseif(RT_ALLOW_VIOLATIONS)
        target_link_libraries(${name} PRIVATE rt_contract_handler)
        catch_discover_tests(${name} TEST_PREFIX "${name}: ")
    else()
        target_link_libraries(${name} PRIVATE rt_contract_handler rt_test_exit_report)
        catch_discover_tests(${name} TEST_PREFIX "${name}: "
            PROPERTIES FAIL_REGULAR_EXPRESSION "rt-audio: [0-9]+ contract violation")
    endif()
endfunction()
